cmake_minimum_required(VERSION 3.22)
project(NexumGate VERSION 1.0.0 LANGUAGES C CXX)
set(NEXUM_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/external/nexum" CACHE PATH "Pinned Nexum shared core")
add_subdirectory("${NEXUM_CORE_DIR}/External/JUCE" JUCE)
juce_add_binary_data(GateBrand HEADER_NAME GateBrand.h NAMESPACE GateBrand SOURCES assets/NEXUM.png)
juce_add_plugin(NexumGate COMPANY_NAME NEXUM BUNDLE_ID com.nexum.gate PLUGIN_MANUFACTURER_CODE Nxum PLUGIN_CODE NxGt FORMATS VST3 Standalone PRODUCT_NAME "Nexum Gate" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE VST3_CATEGORIES Fx Dynamics COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(NexumGate PRIVATE src/Processor.cpp src/Editor.cpp)
target_compile_features(NexumGate PRIVATE cxx_std_17)
target_include_directories(NexumGate PUBLIC "${NEXUM_CORE_DIR}/shared/NexumDesign/include")
target_compile_definitions(NexumGate PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(NexumGate PRIVATE GateBrand juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
add_executable(NexumGateCheck tests/GateCheck.cpp)
target_compile_features(NexumGateCheck PRIVATE cxx_std_17)
target_compile_definitions(NexumGateCheck PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
target_link_libraries(NexumGateCheck PRIVATE NexumGate juce::juce_audio_utils juce::juce_dsp)
if(MSVC)
 target_compile_options(NexumGate PRIVATE /utf-8 /MP1)
 target_compile_options(NexumGateCheck PRIVATE /utf-8 /MP1)
endif()
enable_testing()
add_test(NAME NexumGateChecks COMMAND NexumGateCheck)
set_tests_properties(NexumGateChecks PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")

# Fresh Init session on standalone launch; host/project recall remains unchanged.
target_sources(NexumGate_Standalone PRIVATE "${NEXUM_CORE_DIR}/shared/NexumDesign/include/NexumStandalone.cpp")
target_compile_definitions(NexumGate_Standalone PRIVATE JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)
